cmake_minimum_required(VERSION 3.14)
project(cell_signal_monitor)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(THIRD_PARTY_DIR ${PROJECT_SOURCE_DIR}/third_party)
set(SRC_DIR ${PROJECT_SOURCE_DIR}/src)
set(INCLUDE_DIR ${PROJECT_SOURCE_DIR}/include)
set(EXAMPLES_DIR ${PROJECT_SOURCE_DIR}/examples)

find_package(CURL REQUIRED)

set(STB_INCLUDE_DIR ${THIRD_PARTY_DIR}/stb)
if(NOT EXISTS ${STB_INCLUDE_DIR}/stb_image.h)
    message(FATAL_ERROR "stb_image.h not found in ${STB_INCLUDE_DIR}")
endif()

if(NOT EXISTS ${STB_INCLUDE_DIR}/stb_image.c)
    file(WRITE ${STB_INCLUDE_DIR}/stb_image.c "#define STB_IMAGE_IMPLEMENTATION\n#include \"stb_image.h\"\n")
endif()

add_library(stb_image STATIC ${STB_INCLUDE_DIR}/stb_image.c)
target_include_directories(stb_image PUBLIC ${STB_INCLUDE_DIR})
target_compile_options(stb_image PRIVATE -fPIC)

if(APPLE)
    find_path(GLEW_INCLUDE_DIR GL/glew.h
        PATHS /usr/local/include /opt/homebrew/include /opt/local/include)
    
    find_library(GLEW_LIBRARY NAMES GLEW glew
        PATHS /usr/local/lib /opt/homebrew/lib /opt/local/lib)
    
    if(GLEW_INCLUDE_DIR AND GLEW_LIBRARY)
        set(GLEW_FOUND TRUE)
        set(GLEW_INCLUDE_DIRS ${GLEW_INCLUDE_DIR})
        set(GLEW_LIBRARIES ${GLEW_LIBRARY})
        message(STATUS "Found GLEW: ${GLEW_LIBRARY}")
    else()
        message(FATAL_ERROR "GLEW not found. Install with: brew install glew")
    endif()
else()
    find_package(GLEW REQUIRED)
endif()

find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)

find_path(ZMQ_INCLUDE_DIR zmq.hpp
    PATHS /usr/local/opt/zeromq/include /usr/local/include /opt/homebrew/include)

find_library(ZMQ_LIBRARY NAMES zmq libzmq
    PATHS /usr/local/opt/zeromq/lib /usr/local/lib /opt/homebrew/lib)

if(ZMQ_INCLUDE_DIR AND ZMQ_LIBRARY)
    set(ZMQ_FOUND TRUE)
    set(ZMQ_INCLUDE_DIRS ${ZMQ_INCLUDE_DIR})
    set(ZMQ_LIBRARIES ${ZMQ_LIBRARY})
    message(STATUS "Found ZeroMQ: ${ZMQ_LIBRARY}")
else()
    message(FATAL_ERROR "ZeroMQ not found. Install with: brew install zmq cppzmq")
endif()

find_path(POSTGRES_INCLUDE_DIR pqxx/pqxx
    PATHS /usr/local/include /opt/homebrew/include /usr/local/include/postgresql)

find_library(POSTGRES_LIBRARY NAMES pqxx libpqxx
    PATHS /usr/local/lib /opt/homebrew/lib)

if(POSTGRES_INCLUDE_DIR AND POSTGRES_LIBRARY)
    set(POSTGRES_FOUND TRUE)
    set(POSTGRES_INCLUDE_DIRS ${POSTGRES_INCLUDE_DIR})
    set(POSTGRES_LIBRARIES ${POSTGRES_LIBRARY})
    message(STATUS "Found PostgreSQL: ${POSTGRES_LIBRARY}")
else()
    message(FATAL_ERROR "PostgreSQL (libpqxx) not found. Install with: brew install libpqxx")
endif()

include_directories(${INCLUDE_DIR})
include_directories(${GLEW_INCLUDE_DIRS})
include_directories(${SDL2_INCLUDE_DIRS})
include_directories(${OPENGL_INCLUDE_DIRS})
include_directories(${ZMQ_INCLUDE_DIRS})
include_directories(${POSTGRES_INCLUDE_DIRS})
include_directories(${CURL_INCLUDE_DIRS})
include_directories(${STB_INCLUDE_DIR})

add_library(imgui STATIC
    ${THIRD_PARTY_DIR}/imgui/imgui.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_draw.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_demo.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_tables.cpp
    ${THIRD_PARTY_DIR}/imgui/imgui_widgets.cpp
    ${THIRD_PARTY_DIR}/imgui/backends/imgui_impl_sdl2.cpp
    ${THIRD_PARTY_DIR}/imgui/backends/imgui_impl_opengl3.cpp
)
target_include_directories(imgui PUBLIC ${THIRD_PARTY_DIR}/imgui)
target_compile_options(imgui PRIVATE -fPIC)

add_library(implot STATIC
    ${THIRD_PARTY_DIR}/implot/implot.cpp
    ${THIRD_PARTY_DIR}/implot/implot_demo.cpp
    ${THIRD_PARTY_DIR}/implot/implot_items.cpp
)
target_include_directories(implot PUBLIC ${THIRD_PARTY_DIR}/implot)
target_link_libraries(implot PRIVATE imgui)
target_compile_options(implot PRIVATE -fPIC)

add_library(core STATIC
    ${SRC_DIR}/core/GeoData.cpp
    ${SRC_DIR}/core/DataPoint.cpp
)
target_include_directories(core PUBLIC ${INCLUDE_DIR})
target_compile_options(core PRIVATE -fPIC)

add_library(utils STATIC
    ${SRC_DIR}/utils/Logger.cpp
    ${SRC_DIR}/utils/NetworkUtils.cpp
)
target_include_directories(utils PUBLIC ${INCLUDE_DIR})
target_compile_options(utils PRIVATE -fPIC)

add_library(parser STATIC
    ${SRC_DIR}/parser/JsonParser.cpp
)
target_include_directories(parser PUBLIC ${INCLUDE_DIR})
target_link_libraries(parser PRIVATE core utils)
target_compile_options(parser PRIVATE -fPIC)

add_library(storage STATIC
    ${SRC_DIR}/storage/PostgresStorage.cpp
)
target_include_directories(storage PUBLIC ${INCLUDE_DIR})
target_link_libraries(storage PRIVATE core parser utils ${POSTGRES_LIBRARIES})
target_compile_options(storage PRIVATE -fPIC)

add_library(network STATIC
    ${SRC_DIR}/network/NetworkServer.cpp
)
target_include_directories(network PUBLIC ${INCLUDE_DIR})
target_link_libraries(network PRIVATE core parser storage utils ${ZMQ_LIBRARIES})
target_compile_options(network PRIVATE -fPIC)

add_library(map STATIC
    ${SRC_DIR}/map/MapWindow.cpp
)
target_include_directories(map PUBLIC ${INCLUDE_DIR})
target_link_libraries(map PRIVATE imgui implot stb_image ${CURL_LIBRARIES} ${OPENGL_LIBRARIES})
target_compile_options(map PRIVATE -fPIC)

add_library(gui STATIC
    ${SRC_DIR}/gui/GUIApplication.cpp
)
target_include_directories(gui PUBLIC ${INCLUDE_DIR})
target_link_libraries(gui PRIVATE 
    core storage utils imgui implot map
    ${SDL2_LIBRARIES} ${GLEW_LIBRARIES} ${OPENGL_LIBRARIES}
)
target_compile_options(gui PRIVATE -fPIC)

add_executable(cell_signal_monitor ${SRC_DIR}/main.cpp)

if(APPLE)
    find_library(COCOA_LIBRARY Cocoa)
    find_library(IOKIT_LIBRARY IOKit)
    find_library(COREVIDEO_LIBRARY CoreVideo)
    
    target_link_libraries(cell_signal_monitor PRIVATE 
        gui network storage parser utils core map
        imgui implot stb_image
        ${SDL2_LIBRARIES} ${GLEW_LIBRARIES}
        ${OPENGL_LIBRARIES} ${POSTGRES_LIBRARIES}
        ${COCOA_LIBRARY} ${IOKIT_LIBRARY} ${COREVIDEO_LIBRARY}
        ${ZMQ_LIBRARIES} ${CURL_LIBRARIES} pthread dl
    )
else()
    target_link_libraries(cell_signal_monitor PRIVATE 
        gui network storage parser utils core map
        imgui implot stb_image
        ${SDL2_LIBRARIES} ${GLEW_LIBRARIES}
        ${OPENGL_LIBRARIES} ${POSTGRES_LIBRARIES}
        ${ZMQ_LIBRARIES} ${CURL_LIBRARIES} pthread dl
    )
endif()

target_include_directories(cell_signal_monitor PRIVATE
    ${GLEW_INCLUDE_DIRS} ${SDL2_INCLUDE_DIRS}
    ${INCLUDE_DIR} ${ZMQ_INCLUDE_DIRS} ${POSTGRES_INCLUDE_DIRS}
    ${CURL_INCLUDE_DIRS} ${STB_INCLUDE_DIR}
)

set_target_properties(cell_signal_monitor PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin
)

if(EXISTS ${EXAMPLES_DIR}/osm_tiles/tile_catcher.cpp)
    add_executable(tile_catcher ${EXAMPLES_DIR}/osm_tiles/tile_catcher.cpp)
    target_link_libraries(tile_catcher PRIVATE imgui implot stb_image ${CURL_LIBRARIES} ${SDL2_LIBRARIES} ${OPENGL_LIBRARIES} ${GLEW_LIBRARIES})
    target_include_directories(tile_catcher PRIVATE ${STB_INCLUDE_DIR})
    if(APPLE)
        target_link_libraries(tile_catcher PRIVATE ${COCOA_LIBRARY} ${IOKIT_LIBRARY} ${COREVIDEO_LIBRARY})
    endif()
endif()

message(STATUS "Configuration complete!")
message(STATUS "  CURL: ${CURL_LIBRARIES}")
message(STATUS "  STB Image: ${STB_INCLUDE_DIR}")
